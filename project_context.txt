import streamlit as st
import pandas as pd
import numpy as np
import plotly.express as px

st.set_page_config(page_title="Amazon Dashboard", layout="wide")

@st.cache_data
def load_data():
    file_path = '/Users/hassanmulla/Downloads/NOVA_Phase2/Test/amazon.csv'
    df = pd.read_csv(file_path)
    df['discounted_price_num'] = pd.to_numeric(df['discounted_price'].astype(str).str.replace(r'[^\d.]', '', regex=True), errors='coerce')
    df['actual_price_num'] = pd.to_numeric(df['actual_price'].astype(str).str.replace(r'[^\d.]', '', regex=True), errors='coerce')
    df['discount_pct_num'] = pd.to_numeric(df['discount_percentage'].astype(str).str.replace('%', '', regex=True), errors='coerce')
    df['rating_count_num'] = pd.to_numeric(df['rating_count'].astype(str).str.replace(',', '', regex=True), errors='coerce')
    df['rating_num'] = pd.to_numeric(df['rating'], errors='coerce')
    return df

df = load_data()

st.sidebar.header("Options")
cats = st.sidebar.multiselect("Category", df['category'].unique(), default=df['category'].unique()[:5])
rat_min, rat_max = st.sidebar.slider("Rating", 0.0, 5.0, (0.0, 5.0), 0.1)
p_min = float(df['discounted_price_num'].min())
p_max = float(df['discounted_price_num'].max())
pri_min, pri_max = st.sidebar.slider("Discounted Price", p_min, p_max, (p_min, p_max))

data = df[(df['category'].isin(cats)) & (df['rating_num'] >= rat_min) & (df['rating_num'] <= rat_max) & (df['discounted_price_num'] >= pri_min) & (df['discounted_price_num'] <= pri_max)]

st.title("Amazon Dataset Dashboard")

k1, k2, k3, k4 = st.columns(4)
k1.metric("Products", len(data))
k2.metric("Avg Rating", round(data['rating_num'].mean(), 2))
k3.metric("Avg Discount %", round(data['discount_pct_num'].mean(), 2))
k4.metric("Categories", data['category'].nunique())

c1, c2 = st.columns(2)
top_cats = data['category'].value_counts().head(10).reset_index()
top_cats.columns = ['category', 'count']
fig1 = px.bar(top_cats, x='category', y='count', title="Top 10 Categories")
c1.plotly_chart(fig1, use_container_width=True)

cat_sum = data.groupby('category')['discounted_price_num'].sum().reset_index().sort_values('discounted_price_num', ascending=False).head(10)
fig2 = px.bar(cat_sum, x='category', y='discounted_price_num', title="Top 10 Categories by Value")
c2.plotly_chart(fig2, use_container_width=True)

c3, c4 = st.columns(2)
fig3 = px.histogram(data, x='rating_num', nbins=10, title="Rating Distribution")
c3.plotly_chart(fig3, use_container_width=True)

c4.plotly_chart(fig4, use_container_width=True)

nums = ['actual_price_num', 'discounted_price_num', 'discount_pct_num', 'rating_num', 'rating_count_num']

import pandas as pd
import numpy as np

def load_and_clean(file_path):
    df = pd.read_csv(file_path)
    
    def clean_price(series):
        return pd.to_numeric(series.astype(str).str.replace(r'[^\d.]', '', regex=True), errors='coerce')
    
    def clean_pct(series):
        return pd.to_numeric(series.astype(str).str.replace('%', '', regex=True), errors='coerce')
    
    def clean_count(series):
        return pd.to_numeric(series.astype(str).str.replace(',', '', regex=True), errors='coerce')
    
    df['discounted_price_num'] = clean_price(df['discounted_price'])
    df['actual_price_num'] = clean_price(df['actual_price'])
    df['discount_pct_num'] = clean_pct(df['discount_percentage'])
    df['rating_count_num'] = clean_count(df['rating_count'])
    df['rating_num'] = pd.to_numeric(df['rating'], errors='coerce')
    
    return df

def analyze(file_path):
    df = load_and_clean(file_path)
    
    print("=== DATASET SHAPE ===")
    print(f"Rows: {df.shape[0]} | Columns: {df.shape[1]}\n")
    
    print("=== MISSING VALUES ===")
    print(df.isnull().sum().sort_values(ascending=False))
    print()
    
    numeric_cols = ['actual_price_num', 'discounted_price_num', 'discount_pct_num', 'rating_num', 'rating_count_num']
    
    print("=== DESCRIPTIVE STATISTICS ===")
    print(df[numeric_cols].describe().T)
    print()
    
    print("=== SKEWNESS & KURTOSIS ===")
    print(df[numeric_cols].agg(['skew', 'kurtosis']))
    print()
    
    print("=== CORRELATION MATRIX ===")
    print(df[numeric_cols].corr())
    print()
    
    print("=== TOP 10 CATEGORIES ===")
    print(df['category'].value_counts().head(10))
    print()
    
    print("=== RATING DISTRIBUTION ===")
    print(df['rating_num'].value_counts().sort_index())
    print()
    
    print("=== DUPLICATE PRODUCT_IDS ===")
    print(df['product_id'].duplicated().sum())
    print()
    
    print("=== ZERO/NEGATIVE PRICE ROWS ===")
    print(f"Actual price <= 0: {(df['actual_price_num'] <= 0).sum()}")
    print(f"Discounted price <= 0: {(df['discounted_price_num'] <= 0).sum()}")
    print()
    
    print("=== AVG RATING BY DISCOUNT BUCKET ===")
    df['discount_bucket'] = pd.cut(df['discount_pct_num'], bins=[0, 20, 40, 60, 80, 100])
    print(df.groupby('discount_bucket')['rating_num'].mean())
    print()
    
    return df

# Personal Finance Dashboard

A lightweight, responsive single-page dashboard for visualizing personal finance data. Features a glassmorphism UI, dark/light theme support, and interactive charts.

## File Layout

| File | Purpose |
|------|---------|
| `index.html` | Main page scaffold |
| `style.css` | Global styles and glassmorphism palette |
| `app.js` | Application logic and chart handling |
| `data.json` | Sample finance data |

## Quick Start

Open `index.html` in any modern browser:

```bash
open index.html
```

No build step is required.

## Features

- **Overview Cards**: Monthly balance, total savings, and expenses.
- **Charts**: Expense breakdown (pie), income trend (line), category comparison (bar).
- **Transactions**: View all transactions in a sortable table.
- **Add Transaction**: Modal form to append new entries.
- **Export CSV**: Download transaction data.
- **Theme Toggle**: Persisted dark/light mode.

## Testing Checklist

### Unit Tests
- [ ] Theme toggle updates CSS variables and `localStorage`.
- [ ] Refresh button re-fetches and re-renders data.
- [ ] Export CSV generates a valid downloadable file.
- [ ] Add Transaction form appends to the dataset and updates UI.

### End-to-End Tests
- [ ] Dashboard renders all charts and cards on load.
- [ ] Clicking **Add Transaction** opens the modal.
- [ ] Submitting a new transaction updates the table and charts.
- [ ] Theme change persists after page refresh.

## Build Notes

All assets are static. For future development, data can be swapped from `data.json` to an API endpoint in `app.js`.

(function() {
  const root = document.documentElement;
  const themeToggle = document.getElementById('theme-toggle');
  const refreshBtn = document.getElementById('refresh-btn');
  const exportBtn = document.getElementById('export-btn');
  const addTransactionBtn = document.getElementById('add-transaction-btn');
  const modal = document.getElementById('modal');
  const cancelBtn = document.getElementById('cancel-btn');
  const transactionForm = document.getElementById('transaction-form');
  const txTableBody = document.querySelector('#transaction-table tbody');
  let appData = null;
  let chartInstances = {};
  const STORAGE_KEY = 'pfd_extra_transactions';

  function formatCurrency(value) {
    return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(value);
  }

  function getExtraTransactions() {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      return raw ? JSON.parse(raw) : [];
    } catch {
      return [];
    }
  }

  function saveExtraTransaction(transaction) {
    const extra = getExtraTransactions();
    extra.push(transaction);
    localStorage.setItem(STORAGE_KEY, JSON.stringify(extra));
  }

  async function fetchData() {
    const res = await fetch('data.json');
    const data = await res.json();
    const extra = getExtraTransactions();
    if (!Array.isArray(data.transactions)) data.transactions = [];
    data.transactions = data.transactions.concat(extra);
    return data;
  }

  function computeMetrics(data) {
    const txs = Array.isArray(data.transactions) ? data.transactions : [];
    const income = txs.filter(t => t.amount > 0).reduce((s, t) => s + t.amount, 0);
    const expenses = txs.filter(t => t.amount < 0).reduce((s, t) => s + Math.abs(t.amount), 0);
    const balance = income - expenses;
    const savings = typeof data.savings === 'number' ? data.savings : 0;
    return { income, expenses, balance, savings };
  }

  function renderCards(data) {
    const { income, expenses, balance, savings } = computeMetrics(data);
    document.getElementById('monthly-balance').textContent = formatCurrency(balance);
    document.getElementById('total-savings').textContent = formatCurrency(savings);
    document.getElementById('total-expenses').textContent = formatCurrency(expenses);
    document.getElementById('total-income').textContent = formatCurrency(income);
  }

  function getChartColors() {
    const isDark = root.classList.contains('dark');
    return {
      text: isDark ? '#94a3b8' : '#475569',
      grid: isDark ? 'rgba(255,255,255,0.08)' : 'rgba(0,0,0,0.05)',
      tooltipBg: isDark ? '#1e293b' : '#ffffff',
      tooltipText: isDark ? '#f8fafc' : '#0f172a',
    };
  }

  function destroyChart(key) {
    if (chartInstances[key]) {
      chartInstances[key].destroy();
      delete chartInstances[key];
    }
  }

  function renderCharts(data) {
    const colors = getChartColors();
    Chart.defaults.color = colors.text;
    Chart.defaults.borderColor = colors.grid;

    const expenseData = Array.isArray(data.expenseBreakdown) ? data.expenseBreakdown : [];
    const pieCtx = document.getElementById('expense-pie').getContext('2d');
    destroyChart('pie');
    chartInstances['pie'] = new Chart(pieCtx, {
      type: 'doughnut',
      data: {
        labels: expenseData.map(d => d.category),
        datasets: [{
          data: expenseData.map(d => d.amount),
          backgroundColor: ['#0ea5e9','#8b5cf6','#ec4899','#f59e0b','#10b981','#ef4444'],
          borderWidth: 0
        }]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        plugins: {
          legend: { position: 'bottom' },
          tooltip: {
            backgroundColor: colors.tooltipBg,
            titleColor: colors.tooltipText,
            bodyColor: colors.tooltipText,
            borderColor: colors.grid,
            borderWidth: 1
          }
        }
      }
    });

    const incomeData = Array.isArray(data.monthlyIncome) ? data.monthlyIncome : [];
    const lineCtx = document.getElementById('income-line').getContext('2d');
    destroyChart('line');
    chartInstances['line'] = new Chart(lineCtx, {
      type: 'line',
      data: {
        labels: incomeData.map(d => d.month),
        datasets: [{
          label: 'Income',
          data: incomeData.map(d => d.amount),
          borderColor: '#10b981',
          backgroundColor: 'rgba(16, 185, 129, 0.1)',
          fill: true,
          tension: 0.4,
          pointRadius: 4,
          pointHoverRadius: 6
        }]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        scales: {
          x: { grid: { display: false } },
          y: { beginAtZero: true, grid: { color: colors.grid } }
        },
        plugins: {
          legend: { display: false },
          tooltip: {
            backgroundColor: colors.tooltipBg,
            titleColor: colors.tooltipText,
            bodyColor: colors.tooltipText,
            borderColor: colors.grid,
            borderWidth: 1
          }
        }
      }
    });

    const catData = Array.isArray(data.categoryComparison) ? data.categoryComparison : [];
    const barCtx = document.getElementById('category-bar').getContext('2d');
    destroyChart('bar');
    chartInstances['bar'] = new Chart(barCtx, {
      type: 'bar',
      data: {
        labels: catData.map(d => d.category),
        datasets: [
          {
            label: 'Budgeted',
            data: catData.map(d => d.budgeted),
            backgroundColor: '#0ea5e9',
            borderRadius: 6
          },
          {
            label: 'Actual',
            data: catData.map(d => d.actual),
            backgroundColor: '#ec4899',
            borderRadius: 6
          }
        ]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        scales: {
          x: { grid: { display: false } },
          y: { beginAtZero: true, grid: { color: colors.grid } }
        },
        plugins: {
          legend: { position: 'top' },
          tooltip: {
            backgroundColor: colors.tooltipBg,
            titleColor: colors.tooltipText,
            bodyColor: colors.tooltipText,
            borderColor: colors.grid,
            borderWidth: 1
          }
        }
      }
    });
  }

  function renderTransactions(transactions) {
    txTableBody.innerHTML = '';
    const txs = Array.isArray(transactions) ? transactions : [];
    const sorted = [...txs].sort((a, b) => new Date(b.date) - new Date(a.date));
    sorted.forEach(t => {
      const tr = document.createElement('tr');
      const amountClass = t.amount < 0 ? 'negative' : 'positive';
      tr.innerHTML = `
        <td>${t.date}</td>
        <td>${t.category}</td>
        <td>${t.description}</td>
        <td class="numeric ${amountClass}">${formatCurrency(t.amount)}</td>
      `;
      txTableBody.appendChild(tr);
    });
  }

  function exportCSV() {
    if (!appData || !Array.isArray(appData.transactions)) return;
    const headers = ['Date', 'Category', 'Description', 'Amount'];
    const rows = appData.transactions.map(t => [t.date, t.category, `"${(t.description || '').replace(/"/g, '""')}"`, t.amount]);
    const csv = [
      headers.join(',''),
      ...rows.map(r => r.join(','))
    ].join('\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'transactions.csv';
    a.click();
    URL.revokeObjectURL(url);
  }

  function updateToggleUI() {
    const isDark = root.classList.contains('dark');
    if (themeToggle) {
      themeToggle.innerHTML = isDark
        ? `<span class="toggle-icon">☀️</span> <span class="toggle-text">Light Mode</span>`
        : `<span class="toggle-icon">🌙</span> <span class="toggle-text">Dark Mode</span>`;
    }
  }

  function toggleTheme() {
    const isDark = root.classList.toggle('dark');
    localStorage.setItem('theme', isDark ? 'dark' : 'light');
    updateToggleUI();
    if (appData) {
      renderCharts(appData);
    }
  }

  async function loadData() {
    try {
      appData = await fetchData();
      renderCards(appData);
      renderCharts(appData);
      renderTransactions(appData.transactions);
    } catch (e) {
      console.error('Failed to load data', e);
    }
  }

  function initTheme() {
    const saved = localStorage.getItem('theme') || 'light';
    if (saved === 'dark') root.classList.add('dark');
    updateToggleUI();
  }

  // Event Listeners
  if (themeToggle) themeToggle.addEventListener('click', toggleTheme);
  if (refreshBtn) refreshBtn.addEventListener('click', loadData);
  if (exportBtn) exportBtn.addEventListener('click', exportCSV);

  if (addTransactionBtn) {
    addTransactionBtn.addEventListener('click', () => {
      modal.classList.remove('hidden');
    });
  }

  if (cancelBtn) {
    cancelBtn.addEventListener('click', () => {
      modal.classList.add('hidden');
      transactionForm.reset();
    });
  }

  if (modal) {
    modal.addEventListener('click', (e) => {
      if (e.target === modal) {
        modal.classList.add('hidden');
        transactionForm.reset();
      }
    });
  }

  if (transactionForm) {
    transactionForm.addEventListener('submit', (e) => {
      e.preventDefault();
      const date = document.getElementById('date').value;
      const category = document.getElementById('category').value;
      const description = document.getElementById('description').value;
      const amount = parseFloat(document.getElementById('amount').value);
      if (!date || !category || isNaN(amount)) return;
      const newTx = { date, category, description, amount };
      saveExtraTransaction(newTx);
      if (!appData) appData = { transactions: [] };
      if (!Array.isArray(appData.transactions)) appData.transactions = [];
      appData.transactions.push(newTx);
      renderCards(appData);
      renderTransactions(appData.transactions);
      modal.classList.add('hidden');
      transactionForm.reset();
    });
  }

  initTheme();
  loadData();
})();

import pandas as pd
import numpy as np


def clean_price(series):
    return pd.to_numeric(series.astype(str).str.replace(r'[^\d.]', '', regex=True), errors='coerce')


def clean_pct(series):
    return pd.to_numeric(series.astype(str).str.replace('%', '', regex=True), errors='coerce')


def clean_count(series):
    return pd.to_numeric(series.astype(str).str.replace(',', '', regex=True), errors='coerce')


def load_and_clean(file_path):
    df = pd.read_csv(file_path)
    
    df['discounted_price_num'] = clean_price(df['discounted_price'])
    df['actual_price_num'] = clean_price(df['actual_price'])
    df['discount_pct_num'] = clean_pct(df['discount_percentage'])
    df['rating_count_num'] = clean_count(df['rating_count'])
    df['rating_num'] = pd.to_numeric(df['rating'], errors='coerce')
    df['revenue_proxy'] = df['discounted_price_num'] * df['rating_count_num']
    
    date_candidates = ['date', 'Date', 'timestamp', 'Timestamp', 'sales_date']
    for col in date_candidates:
        if col in df.columns:
            df['parsed_date'] = pd.to_datetime(df[col], errors='coerce')
            break
    else:
        df['parsed_date'] = pd.NaT
    
    if 'country' not in df.columns:
        df['country'] = 'Unknown'
    if 'latitude' not in df.columns:
        df['latitude'] = np.nan
    if 'longitude' not in df.columns:
        df['longitude'] = np.nan
    
    return df
